55// `struct GPIO_TypeDef;` declaration is a hard conflict whichever side is
56// included first.

60// opaque stand-ins so the config structs below still parse; code that actually
61// drives those buses only compiles on the core that enables the module.

64# endif

82# endif
83# ifndef HAL_DAC_MODULE_ENABLED

185// STM32 GPIO PORT/PIN ENCODING
186// ═══════════════════════════════════════════════════════════════════════════════
187// STM32 GPIO is addressed as (Port, Pin) — e.g., PA5 = Port A, Pin 5.
188// We encode this into hf_pin_num_t as: pin_num = (port_index * 16) + pin_index
189// where port_index: A=0, B=1, C=2, ..., K=10.
190// This allows a single hf_pin_num_t to uniquely identify any STM32 GPIO pin.
191// ─────────────────────────────────────────────────────────────────────────────
192
193namespace hf::stm32 {
194
196enum class GpioPort : hf_u8_t {
197 A = 0, B = 1, C = 2, D = 3, E = 4,
198 F = 5, G = 6, H = 7, I = 8, J = 9, K = 10
199};
200
202constexpr hf_pin_num_t EncodePin(GpioPort port, hf_u8_t pin) noexcept {
203 return static_cast<hf_pin_num_t>(static_cast<hf_u8_t>(port) * kPinsPerPort + pin);
204}
205
207constexpr hf_u8_t DecodePinPort(hf_pin_num_t pin) noexcept {
208 return static_cast<hf_u8_t>(pin / kPinsPerPort);
209}
210
212constexpr hf_u8_t DecodePinIndex(hf_pin_num_t pin) noexcept {
213 return static_cast<hf_u8_t>(pin % kPinsPerPort);
214}
215
217constexpr hf_u16_t DecodePinMask(hf_pin_num_t pin) noexcept {
218 return static_cast<hf_u16_t>(1U << DecodePinIndex(pin));
219}

601constexpr HalStatus ToHalStatus(hf_u32_t raw_status) noexcept {

606 case 3: return HalStatus::TIMEOUT;
