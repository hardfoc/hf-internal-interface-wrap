206inline constexpr size_t GetMaxValueSizeForType(hf_nvs_type_t type) noexcept {
207 switch (type) {
208 case hf_nvs_type_t::HF_NVS_TYPE_U8:
209 case hf_nvs_type_t::HF_NVS_TYPE_I8:
210 return 1;
211 case hf_nvs_type_t::HF_NVS_TYPE_U16:
212 case hf_nvs_type_t::HF_NVS_TYPE_I16:
213 return 2;
214 case hf_nvs_type_t::HF_NVS_TYPE_U32:
215 case hf_nvs_type_t::HF_NVS_TYPE_I32:
216 return 4;
217 case hf_nvs_type_t::HF_NVS_TYPE_U64:
218 case hf_nvs_type_t::HF_NVS_TYPE_I64:
219 return 8;
220 case hf_nvs_type_t::HF_NVS_TYPE_STR:
221 case hf_nvs_type_t::HF_NVS_TYPE_BLOB:
222 return HF_NVS_MAX_VALUE_SIZE;
223 default:
224 return 0;
225 }
