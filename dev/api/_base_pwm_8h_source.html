467 static constexpr hf_u32_t DutyCycleToRaw(float duty_cycle, hf_u8_t resolution_bits) noexcept {
468 if (duty_cycle < 0.0f)
469 duty_cycle = 0.0f;
470 if (duty_cycle > 1.0f)
471 duty_cycle = 1.0f;
472 return static_cast<hf_u32_t>(duty_cycle * ((1U << resolution_bits) - 1));
473 }
474
481 static constexpr float RawToDutyCycle(hf_u32_t raw_value, hf_u8_t resolution_bits) noexcept {
482 hf_u32_t max_value = (1U << resolution_bits) - 1;
483 if (raw_value > max_value)
484 raw_value = max_value;
485 return static_cast<float>(raw_value) / static_cast<float>(max_value);
486 }
487
493 static constexpr bool IsValidDutyCycle(float duty_cycle) noexcept {
494 return (duty_cycle >= 0.0f && duty_cycle <= 1.0f);
495 }
496
504 static constexpr bool IsValidFrequency(hf_u32_t frequency_hz, hf_u32_t min_freq_hz,
505 hf_u32_t max_freq_hz) noexcept {
506 return (frequency_hz >= min_freq_hz && frequency_hz <= max_freq_hz);
507 }
508
515 static constexpr bool IsValidRawDuty(hf_u32_t raw_value, hf_u8_t resolution_bits) noexcept {
516 if (resolution_bits == 0 || resolution_bits > 16)
517 return false;
518 hf_u32_t max_value = (1U << resolution_bits) - 1;
519 return (raw_value <= max_value);
520 }
521
528 static constexpr float CalculateFrequencyAccuracy(hf_u32_t target_freq,
529 hf_u32_t actual_freq) noexcept {
530 if (target_freq == 0)
531 return 0.0f;
532 float diff = static_cast<float>(target_freq > actual_freq ? target_freq - actual_freq
533 : actual_freq - target_freq);
534 return 1.0f - (diff / static_cast<float>(target_freq));
535 }
536
542 static constexpr float ClampDutyCycle(float duty_cycle) noexcept {
543 if (duty_cycle < 0.0f)
544 return 0.0f;
545 if (duty_cycle > 1.0f)
546 return 1.0f;
547 return duty_cycle;
548 }
