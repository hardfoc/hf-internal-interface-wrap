377 virtual hf_spi_err_t TransferChain(const hf_u8_t* const* tx_frames, hf_u8_t* const* rx_frames,
378 hf_u16_t frame_length, hf_u16_t frame_count,
379 hf_u32_t inter_frame_gap_us,
380 hf_u32_t timeout_ms = 0) noexcept {
381 if (frame_count == 0U || frame_length == 0U) {
382 return hf_spi_err_t::SPI_ERR_INVALID_PARAMETER;
383 }
384 if (tx_frames == nullptr && rx_frames == nullptr) {
385 return hf_spi_err_t::SPI_ERR_INVALID_PARAMETER;
386 }
387 for (hf_u16_t i = 0; i < frame_count; ++i) {
388 const hf_u8_t* tx = (tx_frames != nullptr) ? tx_frames[i] : nullptr;
389 hf_u8_t* rx = (rx_frames != nullptr) ? rx_frames[i] : nullptr;
390 const hf_spi_err_t err = Transfer(tx, rx, frame_length, timeout_ms);
391 if (err != hf_spi_err_t::SPI_SUCCESS) {
392 return err;
393 }
394 if (inter_frame_gap_us > 0U && (i + 1U) < frame_count) {
395 /* Portable busy-wait; STM32 override uses a tighter spin. */
396 for (hf_u32_t spin = inter_frame_gap_us * 50U; spin > 0U; --spin) {
397 __asm__ volatile("" ::: "memory");
398 }
399 }
400 }
401 return hf_spi_err_t::SPI_SUCCESS;
402 }
