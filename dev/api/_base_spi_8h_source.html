301 virtual hf_spi_err_t TransferChain(const hf_u8_t* const* tx_frames, hf_u8_t* const* rx_frames,
302 hf_u16_t frame_length, hf_u16_t frame_count,
303 hf_u32_t inter_frame_gap_us,
304 hf_u32_t timeout_ms = 0) noexcept {
305 if (frame_count == 0U || frame_length == 0U) {
306 return hf_spi_err_t::SPI_ERR_INVALID_PARAMETER;
307 }
308 if (tx_frames == nullptr && rx_frames == nullptr) {
309 return hf_spi_err_t::SPI_ERR_INVALID_PARAMETER;
310 }
311 for (hf_u16_t i = 0; i < frame_count; ++i) {
312 const hf_u8_t* tx = (tx_frames != nullptr) ? tx_frames[i] : nullptr;
313 hf_u8_t* rx = (rx_frames != nullptr) ? rx_frames[i] : nullptr;
314 const hf_spi_err_t err = Transfer(tx, rx, frame_length, timeout_ms);
315 if (err != hf_spi_err_t::SPI_SUCCESS) {
316 return err;
317 }
318 if (inter_frame_gap_us > 0U && (i + 1U) < frame_count) {
319 /* Portable busy-wait; STM32 override uses a tighter spin. */
320 for (hf_u32_t spin = inter_frame_gap_us * 50U; spin > 0U; --spin) {
321 __asm__ volatile("" ::: "memory");
322 }
323 }
324 }
325 return hf_spi_err_t::SPI_SUCCESS;
326 }
