318constexpr hf_u8_t HfCanDlcToLen(hf_u8_t dlc, bool fd) noexcept {
319 constexpr hf_u8_t kFdTable[16] = {0, 1, 2, 3, 4, 5, 6, 7, 8, 12, 16, 20, 24, 32, 48, 64};
320 if (dlc > 15U) {
321 return 0U;
322 }
323 if (!fd) {
324 return dlc > 8U ? 8U : dlc;
325 }
326 return kFdTable[dlc];
327}

333constexpr hf_u8_t HfCanLenToDlc(hf_u8_t len, bool fd) noexcept {
334 if (len <= 8U) {
335 return len;
336 }
337 if (!fd || len > HF_CAN_FD_MAX_LEN) {
338 return HF_CAN_DLC_INVALID;
339 }
340 for (hf_u8_t dlc = 9U; dlc < 16U; ++dlc) {
341 if (HfCanDlcToLen(dlc, true) >= len) {
342 return dlc;
343 }
344 }
345 return HF_CAN_DLC_INVALID;
346}

352constexpr bool HfCanIsValidLength(hf_u8_t len, bool fd) noexcept {
353 if (!fd) {
354 return len <= HF_CAN_CLASSIC_MAX_LEN;
355 }
356 const hf_u8_t dlc = HfCanLenToDlc(len, true);
357 return dlc != HF_CAN_DLC_INVALID && HfCanDlcToLen(dlc, true) == len;
358}

648 template <typename Frame>
649 void ToFrame(Frame& out) const noexcept {
650 static_assert(sizeof(out.data) == HF_CAN_FD_MAX_LEN, "frame data must be 64 bytes");
651 out.id = id;
652 out.len = dlc;
653 out.flags = GetFlags();
654 for (hf_u8_t i = 0; i < HF_CAN_FD_MAX_LEN; ++i) {
655 out.data[i] = data[i];
656 }
657 out.ts_us = static_cast<decltype(out.ts_us)>(timestamp_us);
658 }
659
665 template <typename Frame>
666 static hf_can_message_t FromFrame(const Frame& in) noexcept {
667 static_assert(sizeof(in.data) == HF_CAN_FD_MAX_LEN, "frame data must be 64 bytes");
668 hf_can_message_t m;
669 m.id = in.id;
670 m.dlc = in.len;
671 (void)m.SetFlags(static_cast<hf_u8_t>(in.flags & hf_can_flag::ALL));
672 for (hf_u8_t i = 0; i < HF_CAN_FD_MAX_LEN; ++i) {
673 m.data[i] = in.data[i];
674 }
675 m.timestamp_us = static_cast<hf_u64_t>(in.ts_us);
676 return m;
677 }
