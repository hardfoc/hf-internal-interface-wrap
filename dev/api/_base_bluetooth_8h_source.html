881inline bool hf_bluetooth_address_t::FromString(const std::string& addr_str) {
882 if (addr_str.length() != 17)
883 return false;
884
885 // Validate format: XX:XX:XX:XX:XX:XX
886 for (size_t i = 0; i < addr_str.length(); ++i) {
887 if (i % 3 == 2) {
888 if (addr_str[i] != ':')
889 return false;
890 } else {
891 if (!std::isxdigit(addr_str[i]))
892 return false;
893 }
894 }
895
896 // Parse using no-exceptions approach with manual hex conversion
897 for (int i = 0; i < 6; i++) {
898 size_t pos = i * 3;
899 const char* hex_start = addr_str.c_str() + pos;
900
901 // Manual hex conversion without exceptions
902 unsigned int value = 0;
903 for (int j = 0; j < 2; j++) {
904 char c = hex_start[j];
905 if (c >= '0' && c <= '9') {
906 value = (value << 4) + (c - '0');
907 } else if (c >= 'A' && c <= 'F') {
908 value = (value << 4) + (c - 'A' + 10);
909 } else if (c >= 'a' && c <= 'f') {
910 value = (value << 4) + (c - 'a' + 10);
911 } else {
912 return false; // Invalid hex character
913 }
914 }
915
916 if (value > 255)
917 return false;
918 addr[i] = static_cast<uint8_t>(value);
919 }
920
921 return true;
922}
