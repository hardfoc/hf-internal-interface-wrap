459 hf_u8_t num_channels, hf_u32_t* readings,
460 float* voltages) noexcept {
461 if (!channel_ids || !readings || !voltages) {
462 return hf_adc_err_t::ADC_ERR_NULL_POINTER;
463 }
464
465 for (hf_u8_t i = 0; i < num_channels; ++i) {
466 hf_adc_err_t err = ReadChannel(channel_ids[i], readings[i], voltages[i]);
467 if (err != hf_adc_err_t::ADC_SUCCESS) {
468 return err;
469 }
470 }
471 return hf_adc_err_t::ADC_SUCCESS;
472 }
