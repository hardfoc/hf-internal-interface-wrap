194constexpr hf_u8_t HfCanDlcToLen(hf_u8_t dlc, bool fd) noexcept {
195 constexpr hf_u8_t kFdTable[16] = {0, 1, 2, 3, 4, 5, 6, 7, 8, 12, 16, 20, 24, 32, 48, 64};
196 if (dlc > 15U) {
197 return 0U;
198 }
199 if (!fd) {
200 return dlc > 8U ? 8U : dlc;
201 }
202 return kFdTable[dlc];
203}
204
209constexpr hf_u8_t HfCanLenToDlc(hf_u8_t len, bool fd) noexcept {
210 if (len <= 8U) {
211 return len;
212 }
213 if (!fd || len > HF_CAN_FD_MAX_LEN) {
214 return HF_CAN_DLC_INVALID;
215 }
216 for (hf_u8_t dlc = 9U; dlc < 16U; ++dlc) {
217 if (HfCanDlcToLen(dlc, true) >= len) {
218 return dlc;
219 }
220 }
221 return HF_CAN_DLC_INVALID;
222}
223
228constexpr bool HfCanIsValidLength(hf_u8_t len, bool fd) noexcept {
229 if (!fd) {
230 return len <= HF_CAN_CLASSIC_MAX_LEN;
231 }
232 const hf_u8_t dlc = HfCanLenToDlc(len, true);
233 return dlc != HF_CAN_DLC_INVALID && HfCanDlcToLen(dlc, true) == len;
234}

478 constexpr bool IsValidId() const noexcept {
479 return id <= (is_extended ? HF_CAN_MAX_EXT_ID : HF_CAN_MAX_STD_ID);

486 constexpr hf_can_frame_error_t Validate() const noexcept {
487 if (!IsValidId()) {
488 return hf_can_frame_error_t::ID_RANGE;
489 }
490 if (is_canfd) {
491 if (is_rtr) {
492 return hf_can_frame_error_t::RTR_ON_FD;
493 }
494 if (dlc > HF_CAN_FD_MAX_LEN) {
495 return hf_can_frame_error_t::FD_TOO_LONG;
496 }
497 if (!HfCanIsValidLength(dlc, true)) {
498 return hf_can_frame_error_t::FD_LENGTH_NOT_EXACT;
499 }
500 return hf_can_frame_error_t::NONE;
501 }
502 if (is_brs || is_esi) {
503 return hf_can_frame_error_t::FD_BITS_ON_CLASSIC;
504 }
505 if (dlc > HF_CAN_CLASSIC_MAX_LEN) {
506 return is_rtr ? hf_can_frame_error_t::RTR_WITH_DATA : hf_can_frame_error_t::CLASSIC_TOO_LONG;
507 }
508 return hf_can_frame_error_t::NONE;
509 }
510
512 constexpr bool IsValid() const noexcept {
513 return Validate() == hf_can_frame_error_t::NONE;

524 template <typename Frame>
525 void ToFrame(Frame& out) const noexcept {
526 static_assert(sizeof(out.data) == HF_CAN_FD_MAX_LEN, "frame data must be 64 bytes");
527 out.id = id;
528 out.len = dlc;
529 out.flags = GetFlags();
530 for (hf_u8_t i = 0; i < HF_CAN_FD_MAX_LEN; ++i) {
531 out.data[i] = data[i];
532 }
533 out.ts_us = static_cast<decltype(out.ts_us)>(timestamp_us);
534 }

541 template <typename Frame>
542 static hf_can_message_t FromFrame(const Frame& in) noexcept {
543 static_assert(sizeof(in.data) == HF_CAN_FD_MAX_LEN, "frame data must be 64 bytes");
544 hf_can_message_t m;
545 m.id = in.id;
546 m.dlc = in.len;
547 (void)m.SetFlags(static_cast<hf_u8_t>(in.flags & hf_can_flag::ALL));
548 for (hf_u8_t i = 0; i < HF_CAN_FD_MAX_LEN; ++i) {
549 m.data[i] = in.data[i];
550 }
551 m.timestamp_us = static_cast<hf_u64_t>(in.ts_us);
552 return m;
553 }
